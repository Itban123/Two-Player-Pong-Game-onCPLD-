module paddle_ball_game_with_scoring(
    input clk,
    input reset,
    input up_p1, down_p1,
    input up_p2, down_p2,
    output reg [7:0] rows,
    output reg [7:0] cols,
    output reg [3:0] score_p1, // Player 1 score
    output reg [3:0] score_p2  // Player 2 score
);

    reg [2:0] p1_pos, p2_pos;        // Paddle positions
    reg [2:0] ball_x, ball_y;        // Ball position
    reg [1:0] ball_speed_x, ball_speed_y; // Ball speed direction
    reg [1:0] display_state;         // Ring counter for multiplexing

    // Clock divider for generating slower display and update clocks
    wire clk_1k, clk_10;
    clock_divider clk_gen1(clk, clk_1k);   // 1 kHz clock for display multiplexing
    clock_divider_10 clk_gen2(clk, clk_10); // 10 Hz clock for game logic updates

    // Initialization block
    initial begin
        rows = 8'b00000000;
        cols = 8'b00000000;       // Active-low columns
        p1_pos = 2;               // Initial position for paddle 1
        p2_pos = 2;               // Initial position for paddle 2
        ball_x = 4;               // Initial position for the ball
        ball_y = 4;
        ball_speed_x = 1;         // Initial ball direction
        ball_speed_y = 1;
        score_p1 = 0;
        score_p2 = 0;
        display_state = 2'b00;    // Start with displaying paddle 1
    end

    // Sequential logic for updating paddle positions and ball mechanics
    always @(posedge clk_10 or posedge reset) begin
        if (reset) begin
            p1_pos <= 2;
            p2_pos <= 2;
            ball_x <= 4;
            ball_y <= 4;
            ball_speed_x <= 1;
            ball_speed_y <= 1;
            score_p1 <= 0;
            score_p2 <= 0;
        end else begin
            // Paddle 1 controls
            if (~up_p1 && p1_pos > 0)  // Move paddle 1 up
                p1_pos <= p1_pos - 1;
            if (down_p1 && p1_pos < 5) // Move paddle 1 down
                p1_pos <= p1_pos + 1;

            // Paddle 2 controls
            if (~up_p2 && p2_pos > 0)  // Move paddle 2 up
                p2_pos <= p2_pos - 1;
            if (down_p2 && p2_pos < 5) // Move paddle 2 down
                p2_pos <= p2_pos + 1;

            // Ball mechanics
            if (ball_x == 0) begin
                if (ball_y >= p1_pos && ball_y <= p1_pos + 2) begin
                    ball_speed_x <= 1; // Reverse direction
                end else begin
                    score_p2 <= score_p2 + 1; // Player 2 scores
                    ball_x <= 4; // Reset ball
                    ball_y <= 4;
                end
            end else if (ball_x == 7) begin
                if (ball_y >= p2_pos && ball_y <= p2_pos + 2) begin
                    ball_speed_x <= 2; // Reverse direction
                end else begin
                    score_p1 <= score_p1 + 1; // Player 1 scores
                    ball_x <= 4; // Reset ball
                    ball_y <= 4;
                end
            end

            // Update ball position
            if (ball_speed_x == 1 && ball_x < 7)
                ball_x <= ball_x + 1;
            else if (ball_speed_x == 2 && ball_x > 0)
                ball_x <= ball_x - 1;

            if (ball_speed_y == 1 && ball_y < 7)
                ball_y <= ball_y + 1;
            else if (ball_speed_y == 2 && ball_y > 0)
                ball_y <= ball_y - 1;

            // Ball boundary checks
            if (ball_y == 7)
                ball_speed_y <= 2;
            else if (ball_y == 0)
                ball_speed_y <= 1;
        end
    end

    // Ring counter for multiplexing display logic
    always @(posedge clk_1k or posedge reset) begin
        if (reset) begin
            display_state <= 2'b00; // Reset display state
        end else begin
            display_state <= display_state + 1; // Cycle through states
        end

        case (display_state)
            2'b00: begin // Display paddle 1
                rows = 8'b00000000;   // Clear rows
                cols = 8'b10000000;   // Active-low for paddle 1
                rows[p1_pos] = 1;
                rows[p1_pos + 1] = 1;
                rows[p1_pos + 2] = 1;
            end
            2'b01: begin // Display paddle 2
                rows = 8'b00000000;   // Clear rows
                cols = 8'b00000001;   // Active-low for paddle 2
                rows[p2_pos] = 1;
                rows[p2_pos + 1] = 1;
                rows[p2_pos + 2] = 1;
            end
            2'b10: begin // Display ball
                rows = 8'b00000000;   // Clear rows
                cols = 8'b00000000;   // Active-low for ball
                rows[ball_x] = 1;
                cols[ball_y] = 1;
            end
            default: begin
                rows = 8'b00000000;
                cols = 8'b00000000;
            end
        endcase
    end

endmodule
